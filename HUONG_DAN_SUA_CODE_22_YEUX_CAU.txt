    def update_work_item(
        item_id: int,
        payload: WorkItemUpdate,
        db: Session = Depends(get_db),
        current_user: User = Depends(get_current_user),
    ):
        work_item = WorkItemService.get_work_item_by_id(db, item_id)
        if not work_item:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail="Hạng mục công việc không tồn tại",
            )

        site = SiteService.get_site_by_id(db, work_item.site_id)
        if not site:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail="Công trình không tồn tại",
            )

        is_owner = site.owner_id == current_user.id
        is_assignee = work_item.assignee_id == current_user.id

        # CHỈ OWNER HOẶC ASSIGNEE MỚI ĐƯỢC PHÉP ĐỔI STATUS
        if payload.status is not None:
            if not (is_owner or is_assignee):
                raise HTTPException(
                    status_code=status.HTTP_403_FORBIDDEN,
                    detail="Chỉ chủ công trình hoặc người được giao việc mới được phép đổi trạng thái",
                )

        # OWNER CÓ THỂ CHỈNH SỬA MỌI TRƯỜNG, NẾU KHÔNG LÀ OWNER CHỈ CHỈNH STATUS
        if not is_owner:
            if payload.assignee_id is not None or payload.title is not None or payload.due_date is not None or payload.priority is not None:
                raise HTTPException(
                    status_code=status.HTTP_403_FORBIDDEN,
                    detail="Chỉ chủ công trình mới có quyền cập nhật tiêu đề, hạn chót, độ ưu tiên hoặc phân công người làm",
                )

        try:
            return WorkItemService.update_work_item(db, item_id, payload)
        except ValueError as e:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail=str(e),
            )

YÊU CẦU 6: Validation due_date phải >= hiện tại + 1 giờ

    from pydantic import BaseModel, ConfigDict, Field, field_validator
    from datetime import datetime, timedelta

    class WorkItemBase(BaseModel):
        title: str = Field(..., min_length=1, max_length=255)
        description: Optional[str] = None
        priority: str = Field(default="MEDIUM", pattern="^(LOW|MEDIUM|HIGH|URGENT)$")
        due_date: Optional[datetime] = None

        @field_validator('due_date')
        @classmethod
        def validate_due_date(cls, v):
            if v is not None:
                now = datetime.utcnow()
                min_due_date = now + timedelta(hours=1)
                if v < min_due_date:
                    raise ValueError("Due date phải lớn hơn thời gian hiện tại ít nhất 1 giờ")
            return v

    @router.get(
        "/construction-sites/{site_id}/work-items/count",
        status_code=status.HTTP_200_OK,
        summary="Đếm tổng số hạng mục và hạng mục đã hoàn thành",
        description="""
    Lấy số lượng tổng hạng mục thi công và số hạng mục đã hoàn thành (DONE) trong công trình.
    """,
        responses={
            200: {"description": "Lấy thống kê thành công"},
            401: {"model": ErrorResponse, "description": "Chưa đăng nhập"},
            403: {"model": ErrorResponse, "description": "Bạn không có quyền truy cập công trình này"},
            404: {"model": ErrorResponse, "description": "Công trình không tồn tại"},
        },
    )
    def get_work_items_count(
        site_id: int,
        db: Session = Depends(get_db),
        current_user: User = Depends(get_current_user),
    ):
        site = SiteService.get_site_by_id(db, site_id)
        if not site:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail="Công trình không tồn tại",
            )

        if site.owner_id != current_user.id:
            member = (
                db.query(SiteMember)
                .filter(
                    SiteMember.site_id == site_id,
                    SiteMember.user_id == current_user.id,
                )
                .first()
            )
            if not member:
                raise HTTPException(
                    status_code=status.HTTP_403_FORBIDDEN,
                    detail="Bạn không có quyền truy cập công trình này",
                )

        total_tasks = db.query(WorkItem).filter(WorkItem.site_id == site_id).count()
        done_tasks = (
            db.query(WorkItem)
            .filter(WorkItem.site_id == site_id, WorkItem.status == "DONE")
            .count()
        )

        return {
            "total_tasks": total_tasks,
            "done_tasks": done_tasks,
        }

    from datetime import datetime

    @staticmethod
    def update_work_item_status(db: Session, item_id: int, new_status: str):
        work_item = db.query(WorkItem).filter(WorkItem.id == item_id).first()
        if not work_item:
            return None

        work_item.status = new_status

        # KHI CHUYỂN SANG DONE, GÁN completed_at = hiện tại
        if new_status == "DONE":
            work_item.completed_at = datetime.utcnow()
        else:
            # NẾU CHUYỂN NGƯỢC LẠI, SET NONE
            work_item.completed_at = None

        db.commit()
        db.refresh(work_item)
        return work_item

    def create_work_item(
        payload: WorkItemCreate,
        site_id: Optional[int] = None,
        db: Session = Depends(get_db),
        current_user: User = Depends(get_current_user),
    ):
        target_site_id = site_id or payload.site_id
        payload.site_id = target_site_id

        site = SiteService.get_site_by_id(db, target_site_id)
        if not site:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail="Công trình không tồn tại",
            )

        if site.owner_id != current_user.id:
            member = (
                db.query(SiteMember)
                .filter(
                    SiteMember.site_id == target_site_id,
                    SiteMember.user_id == current_user.id,
                )
                .first()
            )
            if not member:
                raise HTTPException(
                    status_code=status.HTTP_403_FORBIDDEN,
                    detail="Bạn không có quyền tạo công việc trong công trình này",
                )

        # THÊM PHẦN NÀY: Gán mặc định assignee_id = current_user.id
        if payload.assignee_id is None:
            payload.assignee_id = current_user.id

        # THÊM PHẦN NÀY: Validate assignee_id phải là thành viên
        if payload.assignee_id is not None:
            if payload.assignee_id != site.owner_id:
                assignee_member = (
                    db.query(SiteMember)
                    .filter(
                        SiteMember.site_id == target_site_id,
                        SiteMember.user_id == payload.assignee_id,
                    )
                    .first()
                )
                if not assignee_member:
                    raise HTTPException(
                        status_code=status.HTTP_400_BAD_REQUEST,
                        detail="User này không thuộc chiến dịch",
                    )

        try:
            return WorkItemService.create_work_item(db, payload)
        except ValueError as e:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail=str(e),
            )

            .first()
        )
